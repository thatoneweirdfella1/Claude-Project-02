### Mode Selector (5 Interaction Styles)

| Mode | Setup | Loop Condition | Stop Condition |
|------|-------|--------|--------|
| **Consensus** | Two models, same question | Model A → Model B → rebuttal → Model A | Both agree OR 4 rounds max |
| **Adversarial** | Prosecutor vs. Defender | Model A (attacks) → Model B (defends) → counter | Core weakness identified OR 5 rounds |
| **Socratic** | Questioner vs. Expert | Model A asks → Model B answers → follow-up | User says "stop" OR 3 rounds |
| **Devil's Advocate** | Idea + Critic | Model A proposes → Model B critiques → iterate | Proposal refined OR 4 rounds |
| **Synthesis** | Thesis + Antithesis | Model A (position A) → Model B (position B) → Model A (synthesis) | Both perspectives integrated OR 3 rounds |

## 11.4 IMPLEMENTATION PSEUDOCODE

```python
function multi_ai_mode(original_question, stage_4_answer, mode, model_a, model_b):
    
    # STEP 1: Load dialogue configuration
    config = DIALOGUE_CONFIGS[mode]
    // Returns: {initial_prompts, loop_prompts, stop_condition, max_rounds}
    
    # STEP 2: Initialize dialogue
    dialogue = []
    round = 0
    
    # STEP 3: Get initial response from Model B
    prompt_b = config.initial_prompts["model_b"].format(
        original_question=original_question,
        stage_4_answer=stage_4_answer
    )
    response_b = call_model(model_b, prompt_b)
    dialogue.append({
        "round": 1,
        "speaker": "Model B",
        "content": response_b,
        "model": model_b
    })
    round += 1
    
    # STEP 4: Loop until stop condition
    while not should_stop(dialogue, config, round):
        
        # Model A's turn
        previous_response = dialogue[-1]["content"]
        prompt_a = config.loop_prompts["model_a"].format(
            previous_response=previous_response,
            dialogue_history=format_dialogue(dialogue),
            original_question=original_question
        )
        response_a = call_model(model_a, prompt_a)
        dialogue.append({
            "round": round,
            "speaker": "Model A",
            "content": response_a,
            "model": model_a
        })
        round += 1
        
        # Check stop condition
        if should_stop(dialogue, config, round):
            break
        
        # Model B's turn
        previous_response = dialogue[-1]["content"]
        prompt_b = config.loop_prompts["model_b"].format(
            previous_response=previous_response,
            dialogue_history=format_dialogue(dialogue),
            original_question=original_question
        )
        response_b = call_model(model_b, prompt_b)
        dialogue.append({
            "round": round,
            "speaker": "Model B",
            "content": response_b,
            "model": model_b
        })
        round += 1
    
    # STEP 5: Return dialogue
    return {
        "mode": mode,
        "original_question": original_question,
        "model_a": model_a,
        "model_b": model_b,
        "dialogue": dialogue,
        "rounds_completed": round,
        "stop_reason": config.stop_condition
    }

function should_stop(dialogue, config, round):
    
    # Hard stop: max rounds
    if round >= config.max_rounds:
        return true
    
    # Soft stop: condition met
    if config.mode == "CONSENSUS":
        return detect_agreement(dialogue[-1], dialogue[-2])
    
    if config.mode == "ADVERSARIAL":
        return detect_weakness_identified(dialogue)
    
    if config.mode == "SOCRATIC":
        return detect_deep_understanding(dialogue)
    
    if config.mode == "DEVIL_ADVOCATE":
        return detect_proposal_refined(dialogue)
    
    if config.mode == "SYNTHESIS":
        return detect_synthesis_complete(dialogue)
    
    return false
```

```python
function check_account_tokens(account_id):
    tokens_remaining = get_tokens(account_id)
    
    if tokens_remaining == 0:
        return "DEPLETED"
    elif tokens_remaining < 100:
        return "CRITICAL"  # ~1 more response
    elif tokens_remaining < 500:
        return "LOW"       # ~3-5 more responses
    else:
        return "AVAILABLE"
```

### Auto-Swap Logic

```python
function call_model_with_rotation(current_account, message, dialogue_history):
    
    # STEP 1: Try current account
    tokens_status = check_account_tokens(current_account)
    
    if tokens_status == "AVAILABLE":
        # Use current account
        response = call_model(current_account, message)
        update_tokens(current_account, response.tokens_used)
        return {
            "response": response,
            "account_used": current_account,
            "swapped": false
        }
    
    elif tokens_status in ["DEPLETED", "CRITICAL"]:
        # STEP 2: Find next available account (same provider)
        next_account = find_next_available(current_account.provider)
        
        if next_account is null:
            return error("All {provider} accounts depleted")
        
        # STEP 3: Prepare context transfer
        context_package = {
            "dialogue_history": dialogue_history,
            "original_question": original_question,
            "mode": current_mode,
            "previous_account": current_account,
            "token_swap_reason": tokens_status
        }
        
        # STEP 4: Build system prompt for new account (context transfer)
        transfer_prompt = format_context_transfer(context_package)
        
        full_prompt = transfer_prompt + "\n\n" + message
        
        # STEP 5: Call new account
        response = call_model(next_account, full_prompt)
        update_tokens(next_account, response.tokens_used)
        
        # STEP 6: Notify user
        user_notification = f"🔄 {current_account} tokens depleted. {next_account} taking over."
        
        return {
            "response": response,
            "account_used": next_account,
            "swapped": true,
            "swap_reason": tokens_status,
            "notification": user_notification
        }
```

```python
            return ("WEAKNESS_FOUND", "Core weakness identified")
        if detect_refined_argument(dialogue):
            return ("REFINED", "Argument improved through critique")
        if len(dialogue) >= 5:
            return ("MAX_ROUNDS", "5 rounds completed")
    
    if mode == "SOCRATIC":
        # Stop when understanding is deep enough
        if detect_deep_understanding(dialogue):
            return ("UNDERSTANDING_ACHIEVED", "Topic explored thoroughly")
        if len(dialogue) >= 3:
            return ("MAX_ROUNDS", "3 rounds completed")
    
    if mode == "DEVILS_ADVOCATE":
        # Stop when idea is refined enough
        if detect_refined_idea(dialogue):
            return ("IDEA_REFINED", "Idea improved through iteration")
        if len(dialogue) >= 4:
            return ("MAX_ROUNDS", "4 rounds completed")
    
    if mode == "SYNTHESIS":
        # Stop when synthesis is complete
        if detect_synthesis_complete(dialogue):
            return ("SYNTHESIS_COMPLETE", "Perspectives integrated")
        if len(dialogue) >= 3:
            return ("MAX_ROUNDS", "3 rounds completed")
    
    # Default: keep going
    return (None, None)
```

## 11.5 DETECTION FUNCTIONS (Auto-Stop Logic)

```python
def detect_agreement(last_two_responses):
    """Do both models agree?"""
    # Check for: "you're right", "I agree", "that's fair", "good point"
    agreement_keywords = ["agree", "right", "fair", "good point", "concede"]
    
    if any(keyword in last_two_responses[0].lower() for keyword in agreement_keywords):
        if any(keyword in last_two_responses[1].lower() for keyword in agreement_keywords):
            return True
    return False

def detect_core_weakness(dialogue):
    """Has a fundamental flaw been identified?"""
    # Check if both models acknowledge a critical issue
    # Look for: "fatal flaw", "breaks on", "doesn't work because"
    weakness_indicators = ["fatal", "breaks", "doesn't work", "fail", "invalid"]
    
    recent = dialogue[-2:]
    for response in recent:
        if any(word in response.lower() for word in weakness_indicators):
            return True
    return False

def detect_refined_argument(dialogue):
    """Has argument been improved?"""
    # Compare earlier version to latest
    # Check for: "let me refine", "actually", "revised", "improved"
    refinement_words = ["refined", "improved", "revised", "actually", "better"]
    
    if len(dialogue) >= 2:
        latest = dialogue[-1].lower()
        return any(word in latest for word in refinement_words)
    return False

def detect_deep_understanding(dialogue):
    """Has topic been explored thoroughly?"""
    # Check for meta-commentary: "that's a good question"
    # Or: increased specificity in answers
    understanding_signals = ["good question", "hadn't thought of", "clarifies", "now i see"]
    
    recent = dialogue[-2:]
    for response in recent:
        if any(signal in response.lower() for signal in understanding_signals):
            return True
    return False

def detect_proposal_refined(dialogue):
    """Has proposal been improved through critique?"""
    # Check if proposal changed between iterations
    if len(dialogue) >= 3:
        # Compare proposal in round 1 vs round 2
        earlier = dialogue[0]
        later = dialogue[-1]
        
        # Simple heuristic: later version is longer or has "revised"
        if len(later) > len(earlier) * 1.2:  # 20% longer
            return True
        if "revised" in later.lower() or "improved" in later.lower():
            return True
    return False

def detect_synthesis_complete(dialogue):
    """Have both perspectives been integrated?"""
    # Check for: "what if we", "combines", "integrates", "best of both"
    synthesis_words = ["what if", "combines", "integrates", "both", "synthesis"]
    
    latest = dialogue[-1].lower()
    return sum(word in latest for word in synthesis_words) >= 2
```
